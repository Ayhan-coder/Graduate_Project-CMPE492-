\begin{frame}{4. Çapraz-Aile Hakem Denetimi: Veri Hijyeni ve 7B \texttt{bfloat16} Koşusu}
\scriptsize
\textbf{Problem:} İlk çapraz-aile hakem koşusunda (\texttt{Job 33983}, \texttt{max\_new\_tokens=65}) \texttt{Qwen2.5-7B} modelinin \textbf{270/454 (\%59.5)} maddede JSON'u tamamlayamayıp sessizce \texttt{\{"R":1, "D":1\}} varsayılanına düştüğü saptandı (\texttt{Mistral-7B}: \textbf{454/454 gerçek karar, 0 fallback}).

\begin{block}{Adım 1: Fiziksel Ayrıştırma ve Yeniden Hesaplama (\texttt{recompute\_cross\_family\_clean.py})}
\begin{itemize}
    \item 270 fallback satırı \texttt{labels\_panel\_qwen25\_7b\_a40\_fallback270.csv} dosyasına ayrıldı.
    \item Yalnızca \textbf{184 gerçek Qwen kararı} (\texttt{labels\_panel\_qwen25\_7b\_a40\_clean184.csv}) üzerinden uyum yeniden hesaplandı:
    \begin{itemize}
        \item \textbf{Qwen2.5-7B (184) $\times$ Mistral-7B:} $P_{o,R} = \textbf{\%90.22}$ ($\kappa=0.211$, $TN=3$), $P_{o,D} = \textbf{\%92.39}$ ($\kappa=0.336$, $TN=4$).
        \item \textbf{Qwen2.5-7B (184) $\times$ İnsan A1 ($N=155$):} $P_{o,R} = \%86.45$, $P_{o,D} = \%87.74$, ancak \textbf{$TN = 0$}.
    \end{itemize}
\end{itemize}
\end{block}

\begin{block}{Adım 2: Doğal \texttt{bfloat16} + 192 Token Doğrulama Koşusu (\texttt{Slurm Job 34562})}
\begin{itemize}
    \item \texttt{Qwen2.5-7B-Instruct} 4-bit nicemleme olmadan doğal \texttt{bfloat16} ve \texttt{max\_new\_tokens=192} ile tüm 454 maddede yeniden koşturuldu (\texttt{cross\_family\_judge\_report\_bf16.json}):
    \item \textbf{Parse Fallback:} $270/454 \longrightarrow \textbf{2/454 (\%0.4)}$ seviyesine indi!
    \item \textbf{Sonuç:} Qwen $\times$ Mistral uyumu $R$'de \textbf{\%94.71} ($\kappa=0.175$), $D$'de \textbf{\%88.33} ($\kappa=0.184$) çıktı; fakat İnsan \texttt{A1} ($N=399$) karşısında \textbf{$TN = 0$} kalarak 7B hakemin insan-negatiflerini yakalayamamasının nicemleme değil \textbf{model ölçeği sınırı} olduğu kanıtlandı.
\end{itemize}
\end{block}
\end{frame}
